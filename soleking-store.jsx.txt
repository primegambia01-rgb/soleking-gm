  .checkout-page{padding:32px 20px 60px;}
  .ai-page,.admin-page{padding:32px 20px;}
  .orders-table-header,.order-row{grid-template-columns:1fr 1fr;font-size:12px;}
  .admin-kpis{grid-template-columns:1fr 1fr;}
}
`;

/* ─── HELPERS ─── */
const fmt = n => `D ${Number(n).toLocaleString()}`;
const ref = () => "SK-" + Math.random().toString(36).slice(2,8).toUpperCase();

/* ─── MAIN APP ─── */
export default function SoleKingStore() {
  const [view, setView]       = useState("shop");      // shop | checkout | confirm | ai | admin
  const [cat, setCat]         = useState("All");
  const [cart, setCart]       = useState([]);
  const [cartOpen, setCartOpen] = useState(false);
  const [selected, setSelected] = useState(null);
  const [selSize, setSelSize]   = useState(null);
  const [selColor, setSelColor] = useState(null);
  const [orders, setOrders]     = useState([]);
  const [adminAuth, setAdminAuth] = useState(false);
  const [adminPass, setAdminPass] = useState("");
  const [orderRef, setOrderRef] = useState("");
  const [orderForm, setOrderForm] = useState({ name:"", phone:"", city:"", address:"", payment:"Wave Money", notes:"" });

  // AI Chat
  const [msgs, setMsgs] = useState([{ role:"ai", text:"Salaam! 👋 I'm SoleKing's AI Style Advisor. Tell me your lifestyle, occasion, or budget and I'll recommend the perfect pair for you." }]);
  const [chatInput, setChatInput] = useState("");
  const [aiLoading, setAiLoading] = useState(false);
  const chatRef = useRef(null);

  const filteredProducts = cat === "All" ? PRODUCTS : PRODUCTS.filter(p => p.category === cat);
  const cartCount = cart.reduce((s,i) => s + i.qty, 0);
  const cartTotal = cart.reduce((s,i) => s + i.price * i.qty, 0);
  const revenue = orders.reduce((s,o) => s + o.total, 0);

  useEffect(() => { if (chatRef.current) chatRef.current.scrollTop = chatRef.current.scrollHeight; }, [msgs]);

  const addToCart = (product, size, color) => {
    const key = `${product.id}-${size}-${color}`;
    setCart(prev => {
      const ex = prev.find(i => i.key === key);
      if (ex) return prev.map(i => i.key === key ? { ...i, qty: i.qty+1 } : i);
      return [...prev, { key, id:product.id, name:product.name, emoji:product.emoji, price:product.price, size, color, qty:1 }];
    });
    setSelected(null);
    setCartOpen(true);
  };

  const removeFromCart = key => setCart(prev => prev.filter(i => i.key !== key));

  const handleOrder = () => {
    if (!orderForm.name || !orderForm.phone || !orderForm.city) return;
    const newRef = ref();
    const order = {
      id: newRef, ...orderForm,
      items: [...cart],
      total: cartTotal,
      status: "pending",
      date: new Date().toLocaleDateString("en-GB")
    };
    setOrders(prev => [order, ...prev]);
    setOrderRef(newRef);
    setCart([]);
    setView("confirm");
  };

  const sendAI = async () => {
    if (!chatInput.trim() || aiLoading) return;
    const userMsg = chatInput.trim();
    setChatInput("");
    setMsgs(prev => [...prev, { role:"user", text:userMsg }]);
    setAiLoading(true);

    const systemPrompt = `You are the AI Style Advisor for SoleKing GM — Gambia's first AI-powered men's shoe store. 
Recommend from these products: ${PRODUCTS.map(p => `${p.name} (${p.category}, D${p.price})`).join(", ")}.
Keep replies SHORT (2-4 sentences), warm, and mention specific product names. Use light Gambian market awareness.`;

    try {
      const res = await fetch("https://api.anthropic.com/v1/messages", {
        method:"POST",
        headers:{ "Content-Type":"application/json" },
        body: JSON.stringify({
          model:"claude-sonnet-4-20250514",
          max_tokens:300,
          system: systemPrompt,
          messages:[...msgs.filter(m=>m.role!=="ai" || msgs.indexOf(m)>0).map(m=>({role:m.role==="ai"?"assistant":"user",content:m.text})),{role:"user",content:userMsg}]
        })
      });
      const data = await res.json();
      const text = data.content?.[0]?.text || "Let me find you the perfect pair! Can you tell me your occasion or budget?";
      setMsgs(prev => [...prev, { role:"ai", text }]);
    } catch {
      setMsgs(prev => [...prev, { role:"ai", text:"I'd recommend our Air Force Classic for everyday wear or the Oxford Brogue for formal occasions. What's your budget?" }]);
    }
    setAiLoading(false);
  };

  const updateOrderStatus = (id, status) => {
    setOrders(prev => prev.map(o => o.id === id ? { ...o, status } : o));
  };

  /* ── PRODUCT MODAL ── */
  const ProductModal = () => selected && (
    <div className="modal-bg" onClick={() => setSelected(null)}>
      <div className="modal" onClick={e => e.stopPropagation()}>
        <button className="modal-close" onClick={() => setSelected(null)}>×</button>
        <div className="modal-emoji">{selected.emoji}</div>
        <div className="modal-body">
          <div className="modal-cat">{selected.category}</div>
          <div className="modal-name">{selected.name}</div>
          <div className="modal-desc">{selected.desc}</div>
          <div className="modal-price">{fmt(selected.price)} <small>GMD</small></div>
          <div className="option-label">Select Size (EU)</div>
          <div className="option-chips">
            {selected.sizes.map(s => <button key={s} className={`chip${selSize===s?" selected":""}`} onClick={()=>setSelSize(s)}>{s}</button>)}
          </div>
          <div className="option-label">Select Color</div>
          <div className="option-chips">
            {selected.colors.map(c => <button key={c} className={`chip${selColor===c?" selected":""}`} onClick={()=>setSelColor(c)}>{c}</button>)}
          </div>
          <button className="modal-add-btn" disabled={!selSize||!selColor} onClick={()=>addToCart(selected,selSize,selColor)}>
            {selSize && selColor ? "Add to Cart →" : "Select Size & Color"}
          </button>
        </div>
      </div>
    </div>
  );

  /* ── CART DRAWER ── */
  const CartDrawer = () => (
    <div className={`cart-drawer${cartOpen?" open":""}`}>
      <div className="cart-header">
        <div className="cart-title">Your Cart</div>
        <button className="cart-x" onClick={()=>setCartOpen(false)}>×</button>
      </div>
      <div className="cart-items">
        {cart.length === 0 ? <div className="cart-empty">Your cart is empty.<br/>Start shopping 👟</div>
          : cart.map(item => (
            <div className="cart-item" key={item.key}>
              <div className="ci-emoji">{item.emoji}</div>
              <div>
                <div className="ci-name">{item.name}</div>
                <div className="ci-meta">Size {item.size} · {item.color} · Qty {item.qty}</div>
                <div className="ci-price">{fmt(item.price * item.qty)}</div>
              </div>
              <button className="ci-remove" onClick={()=>removeFromCart(item.key)}>×</button>
            </div>
          ))}
      </div>
      {cart.length > 0 && (
        <div className="cart-footer">
          <div className="cart-total-row">
            <span>Total</span>
            <span className="cart-total-amt">{fmt(cartTotal)} GMD</span>
          </div>
          <button className="cart-checkout-btn" onClick={()=>{setCartOpen(false);setView("checkout");}}>
            Proceed to Checkout →
          </button>
        </div>
      )}
    </div>
  );

  /* ── VIEWS ── */
  const ShopView = () => (
    <>
      <section className="hero-strip">
        <div>
          <h1 className="hero-h">Step Into<br/><em>Royalty</em></h1>
          <p className="hero-sub">Premium men's footwear. Local prices. Delivered to your door across The Gambia — pay with Wave, Orange Money, or cash.</p>
        </div>
        <div className="hero-stats">
          <div className="stat"><div className="stat-num">{PRODUCTS.length}</div><div className="stat-label">Styles</div></div>
          <div className="stat"><div className="stat-num">2–4</div><div className="stat-label">Day Delivery</div></div>
          <div className="stat"><div className="stat-num">D550</div><div className="stat-label">Starting Price</div></div>
        </div>
      </section>
      <div className="filter-bar">
        {CATS.map(c => <button key={c} className={`cat-btn${cat===c?" active":""}`} onClick={()=>setCat(c)}>{c}</button>)}
      </div>
      <div className="shop-main">
        <div className="products-grid">
          {filteredProducts.map(p => {
            const tagClass = p.tag==="Best Seller"?"best":p.tag==="New Arrival"?"new":p.tag==="Trending"?"trending":p.tag==="Budget Pick"?"budget":p.tag==="Premium"?"premium":"";
            return (
              <div className="product-card" key={p.id} onClick={()=>{setSelected(p);setSelSize(null);setSelColor(null);}}>
                <div className="pc-emoji">
                  {p.tag && <span className={`pc-tag ${tagClass}`}>{p.tag}</span>}
                  {p.emoji}
                </div>
                <div className="pc-body">
                  <div className="pc-cat">{p.category}</div>
                  <div className="pc-name">{p.name}</div>
                  <div className="pc-price">{fmt(p.price)} <span>GMD</span></div>
                </div>
                <div className="pc-overlay"><button className="pc-overlay-btn">View Details</button></div>
              </div>
            );
          })}
        </div>
      </div>
    </>
  );

  const CheckoutView = () => (
    <div className="checkout-page">
      <h1 className="checkout-title">Checkout</h1>
      <p className="checkout-sub">Fill in your details. We'll confirm your order via WhatsApp.</p>
      <div className="order-summary">
        <div className="os-title">Order Summary</div>
        {cart.map(i => <div className="os-item" key={i.key}><span>{i.name} (Sz {i.size}, {i.color}) × {i.qty}</span><span style={{color:"var(--g)"}}>{fmt(i.price*i.qty)}</span></div>)}
        <div className="os-total"><span>Total</span><span>{fmt(cartTotal)} GMD</span></div>
      </div>
      <div className="checkout-grid">
        <div className="form-group">
          <label>Full Name *</label>
          <input value={orderForm.name} onChange={e=>setOrderForm(p=>({...p,name:e.target.value}))} placeholder="Your full name"/>
        </div>
        <div className="form-group">
          <label>WhatsApp Number *</label>
          <input value={orderForm.phone} onChange={e=>setOrderForm(p=>({...p,phone:e.target.value}))} placeholder="+220 ..."/>
        </div>
        <div className="form-group">
          <label>City *</label>
          <select value={orderForm.city} onChange={e=>setOrderForm(p=>({...p,city:e.target.value}))}>
            <option value="">Select city</option>
            {CITIES.map(c=><option key={c}>{c}</option>)}
          </select>
        </div>
        <div className="form-group">
          <label>Delivery Address</label>
          <input value={orderForm.address} onChange={e=>setOrderForm(p=>({...p,address:e.target.value}))} placeholder="Street or landmark"/>
        </div>
        <div className="form-group full">
          <label>Payment Method</label>
          <div className="payment-options">
            {PAYMENTS.map(pay=>(
              <div key={pay} className={`pay-option${orderForm.payment===pay?" selected":""}`} onClick={()=>setOrderForm(p=>({...p,payment:pay}))}>
                <div className="pay-name">{pay}</div>
                <div className="pay-desc">{pay==="Cash on Delivery"?"Pay when it arrives":"Mobile payment"}</div>
              </div>
            ))}
          </div>
        </div>
        <div className="form-group full">
          <label>Notes (optional)</label>
          <input value={orderForm.notes} onChange={e=>setOrderForm(p=>({...p,notes:e.target.value}))} placeholder="Any special instructions?"/>
        </div>
      </div>
      <button className="place-btn" disabled={!orderForm.name||!orderForm.phone||!orderForm.city} onClick={handleOrder}>
        Place Order — {fmt(cartTotal)} GMD →
      </button>
    </div>
  );

  const ConfirmView = () => (
    <div className="confirm-page">
      <div className="confirm-icon">✓</div>
      <h1 className="confirm-title">Order Placed!</h1>
      <p className="confirm-sub">Thank you {orderForm.name.split(" ")[0]}. Your order has been received and we'll confirm via WhatsApp within the hour.</p>
      <div className="confirm-ref">
        <div className="confirm-ref-label">Your Order Reference</div>
        <div className="confirm-ref-num">{orderRef}</div>
      </div>
      <button className="whatsapp-btn" onClick={()=>window.open(`https://wa.me/2203524122?text=Hi SoleKing! My order ref is ${orderRef}. Payment: ${orderForm.payment}.`)}>
        📱 Confirm on WhatsApp
      </button>
      <br/><br/>
      <button className="back-btn" onClick={()=>{setView("shop");setOrderForm({name:"",phone:"",city:"",address:"",payment:"Wave Money",notes:""});}}>
        ← Continue Shopping
      </button>
    </div>
  );

  const AIView = () => (
    <div className="ai-page">
      <div className="ai-header">
        <h1 className="ai-title">AI Style <em>Advisor</em></h1>
        <p className="ai-sub">Not sure what to buy? Tell our AI your occasion, budget, or style preference and get a personal recommendation.</p>
      </div>
      <div className="chat-box">
        <div className="chat-messages" ref={chatRef}>
          {msgs.map((m,i) => (
            <div key={i} className={`msg ${m.role}`}>
              <div className="msg-sender">{m.role==="ai"?"SoleKing AI":"You"}</div>
              <div className="msg-bubble">{m.text}</div>
            </div>
          ))}
          {aiLoading && (
            <div className="msg ai">
              <div className="msg-sender">SoleKing AI</div>
              <div className="msg-bubble"><div className="ai-typing"><div className="dot"/><div className="dot"/><div className="dot"/></div></div>
            </div>
          )}
        </div>
        <div className="chat-input-row">
          <input className="chat-input" value={chatInput} onChange={e=>setChatInput(e.target.value)} onKeyDown={e=>e.key==="Enter"&&sendAI()} placeholder="e.g. I need shoes for a wedding, budget D2000..."/>
          <button className="chat-send" onClick={sendAI} disabled={aiLoading||!chatInput.trim()}>Send →</button>
        </div>
      </div>
    </div>
  );

  const AdminView = () => !adminAuth ? (
    <div className="admin-login">
      <h2 className="admin-title">Admin Panel</h2>
      <p className="admin-sub">SoleKing GM internal dashboard</p>
      <input className="admin-input" type="password" placeholder="Enter password" value={adminPass} onChange={e=>setAdminPass(e.target.value)} onKeyDown={e=>e.key==="Enter"&&adminPass===ADMIN_PASS&&setAdminAuth(true)}/>
      <button className="admin-btn" onClick={()=>adminPass===ADMIN_PASS?setAdminAuth(true):alert("Wrong password")}>Access Dashboard</button>
    </div>
  ) : (
    <div className="admin-page">
      <div className="admin-header">
        <div className="admin-name">SoleKing Dashboard</div>
        <button className="admin-logout" onClick={()=>{setAdminAuth(false);setAdminPass("");}}>Logout</button>
      </div>
      <div className="admin-kpis">
        <div className="kpi"><div className="kpi-val">{orders.length}</div><div className="kpi-label">Total Orders</div></div>
        <div className="kpi"><div className="kpi-val">{fmt(revenue)}</div><div className="kpi-label">Revenue (GMD)</div></div>
        <div className="kpi"><div className="kpi-val">{orders.filter(o=>o.status==="pending").length}</div><div className="kpi-label">Pending</div></div>
        <div className="kpi"><div className="kpi-val">{orders.filter(o=>o.status==="delivered").length}</div><div className="kpi-label">Delivered</div></div>
      </div>
      {orders.length === 0 ? (
        <div className="orders-table"><div className="no-orders">No orders yet. Share the store link to start selling! 🚀</div></div>
      ) : (
        <div className="orders-table">
          <div className="orders-table-header">
            <span>Ref</span><span>Customer</span><span>City</span><span>Total</span><span>Payment</span><span>Status</span>
          </div>
          {orders.map(o => (
            <div className="order-row" key={o.id}>
              <span style={{color:"var(--g)",fontFamily:"monospace"}}>{o.id}</span>
              <div><div style={{fontWeight:500}}>{o.name}</div><div style={{fontSize:"11px",color:"var(--mu)"}}>{o.phone}</div></div>
              <span>{o.city}</span>
              <span style={{color:"var(--g)",fontWeight:600}}>{fmt(o.total)}</span>
              <span style={{fontSize:"12px"}}>{o.payment}</span>
              <select className="status-select" value={o.status} onChange={e=>updateOrderStatus(o.id,e.target.value)}>
                <option value="pending">Pending</option>
                <option value="confirmed">Confirmed</option>
                <option value="delivered">Delivered</option>
              </select>
            </div>
          ))}
        </div>
      )}
    </div>
  );

  return (
    <>
      <style>{CSS}</style>
      <div>
        {/* NAV */}
        <nav>
          <div className="nav-logo" onClick={()=>setView("shop")}>SoleKing</div>
          <div className="nav-tabs">
            {[["shop","Shop"],["ai","AI Advisor"],["admin","Admin"]].map(([v,l])=>(
              <button key={v} className={`nav-tab${view===v?" active":""}`} onClick={()=>setView(v)}>{l}</button>
            ))}
          </div>
          <div className="nav-right">
            <button className="cart-btn" onClick={()=>setCartOpen(o=>!o)}>
              Cart {cartCount > 0 && <span className="cart-badge">{cartCount}</span>}
            </button>
          </div>
        </nav>

        {/* VIEWS */}
        {view==="shop" && <ShopView/>}
        {view==="checkout" && (cart.length>0 ? <CheckoutView/> : <ShopView/>)}
        {view==="confirm" && <ConfirmView/>}
        {view==="ai" && <AIView/>}
        {view==="admin" && <AdminView/>}

        {/* CART + MODAL */}
        <CartDrawer/>
        <ProductModal/>

        {/* Cart overlay */}
        {cartOpen && <div style={{position:"fixed",inset:0,background:"rgba(0,0,0,.4)",zIndex:599}} onClick={()=>setCartOpen(false)}/>}
      </div>
    </>
  );
}
